// 全局速度控制相关参数
const double MAX_LINEAR_SPEED = 3.0;    // 最大线速度 (m/s)，可按需调整
const double WAYPOINT_TOLERANCE = 0.5;  // 航点到达阈值 (m)
int current_waypoint_idx = 0;           // 当前目标航点索引
bool velocity_control_start = false;    // 速度控制启动标志

// 速度指令发布器
ros::Publisher vel_cmd_publisher;

{
    ros::init(argc, argv, "path_sender"); 
    ros::NodeHandle n; 

    PathSender go(&n);

    return 0;
}

std::vector<std::vector<geometry_msgs::Point>> loadPathsFromYAML(const std::string &filename)
{
    std::vector<std::vector<geometry_msgs::Point>> paths;

    try {
        YAML::Node config = YAML::LoadFile(filename);
        if (!config["paths"]) {
            ROS_ERROR("YAML file does not contain 'paths' key.");
            return paths;
        }

        for (const auto &path_node : config["paths"]) {
            std::vector<geometry_msgs::Point> path;
            for (const auto &point_node : path_node.second) {
                if (point_node.size() != 3) {
                    ROS_WARN("Invalid point format, skipping.");
                    continue;
                }

                geometry_msgs::Point point;
                point.x = point_node[0].as<double>();
                point.y = point_node[1].as<double>();
                point.z = point_node[2].as<double>();

                path.push_back(point);
            }
            paths.push_back(path);
        }
    } catch (const YAML::Exception &e) {
        ROS_ERROR("Error loading YAML file: %s", e.what());
    }

    return paths;
}

void PathSender::POintSet()
{
    for (int i = 0; i < 13; ++i)
    {
        station[i].x = station_[i].x;
        station[i].y = station_[i].y;
        station[i].z = station_[i].z;

        Transit_hub[i].x = Transit_hub_[i].x;
        Transit_hub[i].y = Transit_hub_[i].y;
        Transit_hub[i].z = Transit_hub_[i].z;

        end_point[i].x = end_point_[i].x;
        end_point[i].y = end_point_[i].y;
        end_point[i].z = end_point_[i].z;
    }
}

// 核心：计算当前位置到目标航点的速度指令
void calculate_velocity_cmd(const geometry_msgs::PoseStamped::ConstPtr& current_pose, 
                            const std::vector<geometry_msgs::Point>& full_path)
{
    if (!velocity_control_start || current_waypoint_idx >= full_path.size()) {
        // 未启动/航点完成，发布停止指令
        geometry_msgs::Twist stop_cmd;
        vel_cmd_publisher.publish(stop_cmd);
        if (current_waypoint_idx >= full_path.size()) {
            ROS_INFO("All waypoints reached! Stop moving.");
            velocity_control_start = false;
            current_waypoint_idx = 0;
        }
        return;
    }

    // 当前无人机位置（直接使用，无需改y/z）
    double curr_x = current_pose->pose.position.x;
    double curr_y = current_pose->pose.position.y;
    double curr_z = current_pose->pose.position.z;

    // 目标航点
    geometry_msgs::Point target = full_path[current_waypoint_idx];
    double dx = target.x - curr_x;
    double dy = target.y - curr_y;
    double dz = target.z - curr_z;

    // 计算到目标航点的距离
    double dist_to_target = sqrt(dx*dx + dy*dy + dz*dz);

    // 到达当前航点，切换下一个
    if (dist_to_target < WAYPOINT_TOLERANCE) {
        current_waypoint_idx++;
        ROS_INFO("Reached waypoint %d, switch to next.", current_waypoint_idx);
        return;
    }

    // 生成速度指令（沿航点方向以最大速度飞行）
    geometry_msgs::Twist vel_cmd;
    vel_cmd.linear.x = (dx / dist_to_target) * MAX_LINEAR_SPEED;
    vel_cmd.linear.y = (dy / dist_to_target) * MAX_LINEAR_SPEED;
    vel_cmd.linear.z = (dz / dist_to_target) * MAX_LINEAR_SPEED;
    // 角速度设为0（保持当前航向，如需转向可额外计算）
    vel_cmd.angular.x = 0.0;
    vel_cmd.angular.y = 0.0;
    vel_cmd.angular.z = 0.0;

    // 发布速度指令
    vel_cmd_publisher.publish(vel_cmd);
}

PathSender::PathSender(ros::NodeHandle *nh)
{  
    POintSet();
    paths = loadPathsFromYAML(std::string("/home/m0nesy/basic_dev/src/path_sender/config/paths.yaml"));

    // 1. 订阅无人机状态（用于速度规划）
    initial_pose_suber = nh->subscribe<geometry_msgs::PoseStamped>("/airsim_node/initial_pose", 1, 
                            std::bind(&PathSender::initial_pose_cb, this, std::placeholders::_1));
    end_pose_suber = nh->subscribe<geometry_msgs::PoseStamped>("/airsim_node/end_goal", 1, 
                            std::bind(&PathSender::end_pose_cb, this, std::placeholders::_1));
    // 订阅GPS位姿（核心：用于实时计算速度指令）
    gps_pose_suber = nh->subscribe<geometry_msgs::PoseStamped>("/airsim_node/drone_1/gps", 1, 
                            [this](const geometry_msgs::PoseStamped::ConstPtr& msg) {
                                if (velocity_control_start) {
                                    calculate_velocity_cmd(msg, this->path);
                                }
                                this->gps_pose_cb(msg); // 保留原有GPS处理逻辑
                            });

    // 2. 发布器初始化
    timer = nh->createTimer(ros::Duration(1.0),&PathSender::timeCB,this);
    waypoint_publisher = nh->advertise<path_sender::WayPoints>("/waypoints", 1);
    edited_gps_publisher = nh->advertise<geometry_msgs::PoseWithCovarianceStamped>("/airsim_node/drone_1/edited_gps", 1);
    // 新增：速度指令发布器（话题名适配PX4/AirSim）
    vel_cmd_publisher = nh->advertise<geometry_msgs::Twist>("/airsim_node/drone_1/vel_cmd_body_frame", 10);

    ros::spin();
}

PathSender::~PathSender()
{
}

void PathSender::initial_pose_cb(const geometry_msgs::PoseStamped::ConstPtr& msg)
{ 
    if(!initial_num_get)
    for(int i=1;i<=12;i++)
    {
        if(abs(msg->pose.position.x - station[i].x) <5)
        {
            initial_num = i;
            initial_num_get = true;
            break;
        }
    }    
}

void PathSender::end_pose_cb(const geometry_msgs::PoseStamped::ConstPtr& msg)
{
    if(!end_num_get)
    for(int i=1;i<=12;i++)
    {
        if(abs(msg->pose.position.x - station[i].x) <5)
        {
            end_num = i;
            end_num_get = true;
            break;
        }
    }
}

void PathSender::gps_pose_cb(const geometry_msgs::PoseStamped::ConstPtr& msg)
{
    geometry_msgs::PoseWithCovarianceStamped edited_gps;
    edited_gps.header.stamp = ros::Time::now();
    edited_gps.header.frame_id = "map";
    edited_gps.pose.pose = msg->pose;

    edited_gps.pose.covariance[0] = 1e-7;   // x
    edited_gps.pose.covariance[7] = 1e-7;   // y
    edited_gps.pose.covariance[14] = 1e-7;  // z
    edited_gps.pose.covariance[21] = 0.1;   // roll
    edited_gps.pose.covariance[28] = 0.1;   // pitch
    edited_gps.pose.covariance[35] = 0.1;   // yaw

    edited_gps_publisher.publish(edited_gps);
}

void PathSender::timeCB(const ros::TimerEvent& event)
{
    if(!path_get)
    {
        if(initial_num_get && end_num_get)
        {
            // 拼接核心航点（保留原有逻辑，删除冗余）
            path = paths[initial_num-1];
            path.emplace_back(Transit_hub[initial_num]);
            path.emplace_back(Transit_hub[end_num]);
            std::vector<geometry_msgs::Point> temp_path = paths[end_num-1];
            std::reverse(temp_path.begin(), temp_path.end());
            path.insert(path.end(), temp_path.begin(), temp_path.end());

            path_get = true;
            velocity_control_start = true; // 启动速度控制
            ROS_INFO("Path loaded! Start velocity control. Max speed: %.1f m/s", MAX_LINEAR_SPEED);
        }
    }
    else
    {
        // 保留航点发布（兼容原有规划模块）
        path_sender::WayPoints path_msg;
        path_msg.points = this->path;
        waypoint_publisher.publish(path_msg);
    }
}
